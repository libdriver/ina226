47static void a_receive_callback(uint8_t type)
48{
49 switch (type)
50 {
51 case INA226_STATUS_SHUNT_VOLTAGE_OVER_VOLTAGE :
52 {
53 gs_flag |= 1 << 0;
54 ina226_interface_debug_print("ina226: irq shunt voltage over voltage.\n");
55
56 break;
57 }
58 case INA226_STATUS_SHUNT_VOLTAGE_UNDER_VOLTAGE :
59 {
60 gs_flag |= 1 << 1;
61 ina226_interface_debug_print("ina226: irq shunt voltage under voltage.\n");
62
63 break;
64 }
65 case INA226_STATUS_BUS_VOLTAGE_OVER_VOLTAGE :
66 {
67 gs_flag |= 1 << 2;
68 ina226_interface_debug_print("ina226: irq bus voltage over voltage.\n");
69
70 break;
71 }
72 case INA226_STATUS_BUS_VOLTAGE_UNDER_VOLTAGE :
73 {
74 gs_flag |= 1 << 3;
75 ina226_interface_debug_print("ina226: irq bus voltage under voltage.\n");
76
77 break;
78 }
79 case INA226_STATUS_POWER_OVER_LIMIT :
80 {
81 gs_flag |= 1 << 4;
82 ina226_interface_debug_print("ina226: irq power over limit.\n");
83
84 break;
85 }
86 default :
87 {
88 ina226_interface_debug_print("ina226: unknown code.\n");
89
90 break;
91 }
92 }
93}
